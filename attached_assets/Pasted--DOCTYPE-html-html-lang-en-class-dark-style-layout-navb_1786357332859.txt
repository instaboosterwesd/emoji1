        let isTokenVisible = false;

        function toggleApiTokenVisibility() {
            const displaySpan = document.getElementById("apiTokenDisplay");
            const realToken = document.getElementById("realApiToken").value;
            const toggleIcon = document.getElementById("toggleApiIcon");

            if (!realToken) {
                showToast("⚠️ GENERATE TOKEN FIRST!", "#ff4d4d");
                return;
            }

            if (isTokenVisible) {
                displaySpan.innerText = "••••••••••••••••••••••••••••••••";
                toggleIcon.className = "fa-solid fa-eye";
                isTokenVisible = false;
            } else {
                displaySpan.innerText = realToken;
                toggleIcon.className = "fa-solid fa-eye-slash";
                isTokenVisible = true;
            }
        }

        function toggleCodeApiKeyVisibility() {
            const codeSpan = document.getElementById("codeApiKeySpan");
            const realToken = document.getElementById("realApiToken").value;
            const icon = document.getElementById("toggleCodeKeyIcon");
            const text = document.getElementById("toggleCodeKeyText");

            if (!realToken) {
                showToast("⚠️ GENERATE TOKEN FIRST!", "#ff4d4d");

        function copyApiTokenToClipboard() {
            const realToken = document.getElementById("realApiToken").value;
            if (!realToken) {
                showToast("⚠️ GENERATE TOKEN FIRST!", "#ff4d4d");
                return;
            }
            
            navigator.clipboard.writeText(realToken).then(() => {
                showToast("🚀 COPIED!", "#00f2fe");
            }).catch(() => {
                showToast("⚠️ COPY FAILED!", "#ff4d4d");
            });
        }

        function copyFullCodeWithRealToken() {
            const realToken = document.getElementById("realApiToken").value || 'YOUR_API_KEY';
            const codeBlock = document.getElementById("developerUnifiedCodeBlock").innerText;
            
            // Replaces stars or whatever token text with actual real API Token
            const codeSpanText = document.getElementById("codeApiKeySpan").innerText;
            const fullRealCode = codeBlock.replace(codeSpanText, realToken);

            navigator.clipboard.writeText(fullRealCode).then(() => {
                showToast("🚀 CODE COPIED!", "#00f2fe");
            }).catch(() => {
                showToast("⚠️ COPY FAILED!", "#ff4d4d");
            });
        }

        function showToast(message, color) {
            const toast = document.getElementById("copyToast");
            toast.innerText = message;
            toast.style.borderColor = color;
            toast.style.color = color;
            toast.className = "show";
            
            setTimeout(function(){ 
                toast.className = toast.className.replace("show", ""); 
            }, 1200);
        }
